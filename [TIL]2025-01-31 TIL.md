import re
import random
from datetime import datetime, timedelta
from faker import Faker

fake = Faker()

def parse_create_table(sql: str):
    """
    여러 CREATE TABLE 구문이 포함된 문자열에서 테이블 이름과 컬럼 정보를 파싱합니다.
    반환형: { 테이블명: [(컬럼명, 데이터타입, 타입파라미터), ...] }
    """
    table_pattern = r'CREATE TABLE\s+"?(\w+)"?\s*\((.*?)\);'
    tables = {}
    for table_match in re.finditer(table_pattern, sql, re.DOTALL | re.IGNORECASE):
        table_name = table_match.group(1)
        columns_str = table_match.group(2)
        columns = []
        for line in columns_str.splitlines():
            line = line.strip().rstrip(',')
            if not line:
                continue
            # 단순히 "컬럼명", 데이터타입, (선택적) 파라미터를 추출
            m = re.match(r'"?(\w+)"?\s+([A-Z]+)(\([^)]*\))?', line, re.IGNORECASE)
            if m:
                col_name = m.group(1)
                col_type = m.group(2).upper()
                type_params = m.group(3)  # 예: (20) 또는 (16,2)
                columns.append((col_name, col_type, type_params))
        tables[table_name] = columns
    return tables

def generate_value(col_type: str, type_params: str):
    """
    컬럼의 데이터타입에 따라 임의의 값을 생성합니다.
    (FK로 지정된 컬럼은 이 함수가 호출되지 않고 참조 테이블의 값이 할당됩니다.)
    """
    if col_type == 'NUMBER':
        if type_params:
            nums = list(map(int, re.findall(r'\d+', type_params)))
            if len(nums) == 1:
                precision = nums[0]
                max_val = 10 ** precision - 1
                return random.randint(0, max_val)
            elif len(nums) == 2:
                precision, scale = nums
                max_int_part = 10 ** (precision - scale) - 1
                value = random.uniform(0, max_int_part)
                return round(value, scale)
        else:
            return random.randint(0, 1000)
    
    elif col_type.startswith('VARCHAR'):
        max_length = 10  # 기본 길이
        if type_params:
            nums = re.findall(r'\d+', type_params)
            if nums:
                max_length = int(nums[0])
        word = fake.word()
        while len(word) < max_length // 2:
            word += fake.word()
        return word[:max_length]
    
    elif col_type == 'DATE':
        start_date = datetime(2000, 1, 1)
        end_date = datetime(2025, 12, 31)
        delta_days = (end_date - start_date).days
        random_date = start_date + timedelta(days=random.randint(0, delta_days))
        return random_date.strftime('%Y-%m-%d')
    
    else:
        return None

def generate_dummy_data_with_fk(sql: str, num_rows: int = 5):
    """
    FK 관계를 고려하여 더미 데이터를 생성합니다.
    반환형: { 테이블명: [ {컬럼: 값, ...}, ... ] }
    """
    # 1. 테이블 정의 파싱
    table_defs = parse_create_table(sql)

    # 2. 의존성 순서 (참조 대상 테이블은 먼저 생성)
    dependency_order = [
        "USERS",       # 독립 테이블
        "TICKERS",     # 독립 테이블
        "PRODUCTS",    # 독립 테이블
        "ACCOUNTS",    # USERS 참조 (USER_ID)
        "FUNDS",       # PRODUCTS, TICKERS 참조
        "HOLDINGS",    # ACCOUNTS, PRODUCTS, TICKERS 참조
        "PRICES",      # TICKERS 참조
        "BONDS",       # PRODUCTS 참조
        "TRANSACTIONS",# ACCOUNTS, PRODUCTS 참조
        "TRADES",      # TRANSACTIONS의 (TRANSACTION_ID, DATE) 복합 참조 + ACCOUNTS 참조
        "COMPANIES"    # TICKERS 참조
    ]

    # 3. 개별 외래키 매핑 (테이블: {컬럼: (참조테이블, 참조컬럼)})
    fk_mapping = {
        "ACCOUNTS": {"USER_ID": ("USERS", "USER_ID")},
        "FUNDS": {"PRODUCT_ID": ("PRODUCTS", "PRODUCT_ID"),
                  "STOCK_TICKER": ("TICKERS", "STOCK_TICKER")},
        "HOLDINGS": {"ACCOUNT_ID": ("ACCOUNTS", "ACCOUNT_ID"),
                     "PRODUCT_ID": ("PRODUCTS", "PRODUCT_ID"),
                     "STOCK_TICKER": ("TICKERS", "STOCK_TICKER")},
        "PRICES": {"STOCK_TICKER": ("TICKERS", "STOCK_TICKER")},
        "TRANSACTIONS": {"ACCOUNT_ID": ("ACCOUNTS", "ACCOUNT_ID"),
                         "PRODUCT_ID": ("PRODUCTS", "PRODUCT_ID")},
        "TRADES": {"BUYER_ACCOUNT_NUMBER": ("ACCOUNTS", "ACCOUNT_ID"),
                   "SELLER_ACCOUNT_NUMBER": ("ACCOUNTS", "ACCOUNT_ID")},
        "BONDS": {"PRODUCT_ID": ("PRODUCTS", "PRODUCT_ID")},
        "COMPANIES": {"STOCK_TICKER": ("TICKERS", "STOCK_TICKER")}
    }
    # 4. 복합 외래키 매핑 (예: TRADES의 TRANSACTION_ID와 DATE는 함께 TRANSACTIONS에서 가져옴)
    composite_fk_mapping = {
        "TRADES": {("TRANSACTION_ID", "DATE"): ("TRANSACTIONS", ("TRANSACTION_ID", "DATE"))}
    }

    # 5. 각 테이블별 생성된 데이터를 저장할 딕셔너리
    generated_data = {}

    # 6. 의존성 순서대로 더미 데이터 생성
    for table in dependency_order:
        generated_data[table] = []
        columns = table_defs.get(table, [])
        for _ in range(num_rows):
            row = {}

            # (가) 복합 외래키 처리 (예: TRADES의 TRANSACTION_ID와 DATE)
            if table in composite_fk_mapping:
                for local_cols, (ref_table, ref_cols) in composite_fk_mapping[table].items():
                    ref_row = random.choice(generated_data[ref_table])
                    for local_col, ref_col in zip(local_cols, ref_cols):
                        row[local_col] = ref_row[ref_col]

            # (나) 개별 컬럼에 대해 값 생성
            for col_name, col_type, type_params in columns:
                if col_name in row:
                    continue
                if table in fk_mapping and col_name in fk_mapping[table]:
                    ref_table, ref_col = fk_mapping[table][col_name]
                    if generated_data.get(ref_table):
                        ref_row = random.choice(generated_data[ref_table])
                        row[col_name] = ref_row[ref_col]
                    else:
                        row[col_name] = generate_value(col_type, type_params)
                else:
                    row[col_name] = generate_value(col_type, type_params)

            generated_data[table].append(row)

    return generated_data

def generate_union_all_insert_sql(generated_data: dict):
    """
    생성된 더미 데이터를 바탕으로 각 테이블별로 하나의 INSERT 문에 UNION ALL로 모든 행을 포함시키는 SQL 문을 생성합니다.
    """
    sql_statements = []
    for table, rows in generated_data.items():
        if not rows:
            continue
        # 각 테이블의 첫 행에서 컬럼 순서를 결정합니다.
        columns = list(rows[0].keys())
        columns_sql = ", ".join([f'"{col}"' for col in columns])
        select_statements = []
        for row in rows:
            values_sql_list = []
            for col in columns:
                value = row[col]
                if value is None:
                    values_sql_list.append("NULL")
                elif isinstance(value, (int, float)):
                    values_sql_list.append(str(value))
                else:
                    escaped_value = str(value).replace("'", "''")
                    values_sql_list.append(f"'{escaped_value}'")
            select_statements.append("SELECT " + ", ".join(values_sql_list))
        full_sql = f'INSERT INTO "{table}" ({columns_sql})\n' + " FROM DUAL UNION ALL\n".join(select_statements)
        full_sql += " FROM DUAL;\nCOMMIT;\n"
        sql_statements.append(full_sql)
    return sql_statements

def write_sql_file(sql_statements: list, filename: str):
    """
    생성된 SQL 문들을 파일로 저장합니다.
    """
    with open(filename, 'w', encoding='utf-8') as f:
        for stmt in sql_statements:
            f.write(stmt + "\n\n")
    print(f"SQL 파일 '{filename}'가 생성되었습니다.")
