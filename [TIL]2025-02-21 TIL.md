### 좋아요 가져오기

상세 페이지로 들어갈 때 유저가 좋아요를 눌렀는지 ajax로 구현해서 좋아요를 눌렀다면 취소가 좋아요를 누르지 않았다면 좋아요가 뜨도록 했다. 첫 화면을 만들 때 문제가 발생했다. ajax가 포함된 파일보다 해당 페이지가 열리는게 우선이었다. 그래서 ajax와 페이지를 열기 위해 req를 날리는 순간에 각각 좋아요 여부와 총 좋아요 수를 포함해서 날렸다. ajax를 잘 관리하면 그럴 필요가 없었을 텐데 아쉬움이 남는다.

```java
@Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
//        id 로 게시글 조회
        String id = request.getParameter("id");
        BoardPostVO post = boardDAO.selectPostById(id);
        request.setAttribute("post", post);
        boardDAO.updateViewCount(id);

        // 세션에서 유저 아이디 가져오기
        HttpSession session = request.getSession();
        MemberVO member = (MemberVO) session.getAttribute("member");
        String user = member.getId();

        boolean liked = likeDAO.isLiked(id, user);
        String newAction = liked ? "취소" : "좋아요";
        int likeCount = likeDAO.getLikeCount(id);
        request.setAttribute("newAction", newAction);
        request.setAttribute("likeCount", likeCount);
        request.getRequestDispatcher("/WEB-INF/jsp/viewBoard.jsp").forward(request, response);
    }
```

```java
$(document).ready(function(){
            // 좋아요 버튼 클릭 이벤트
            $("#likeButton").click(function(){
                var postId = $("#postId").val();
                var currentAction = $(this).text().trim(); // "좋아요" 또는 "취소"
                $.ajax({
                    url: "${pageContext.request.contextPath}/toggleLike",
                    type: "POST",
                    data: { id: postId, action: currentAction },
                    success: function(response){
                        $("#likeCount").text(response.likeCount);
                        $("#likeButton").text(response.newAction);
                    },
                    error: function(xhr, status, error){
                        alert("좋아요 처리 중 오류가 발생했습니다.");
                    }
                });
            });
```

### 댓글 가져오기

```java
@Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String id = request.getParameter("id");
        List<CommentVO> comments = commentDAO.selectAllCommentsById(id);
        request.setAttribute("comments", comments);
        response.setContentType("text/html;charset=UTF-8");

        // 전체 페이지가 아니라, 댓글 목록에 해당하는 JSP 조각만 포함시킵니다.
        RequestDispatcher rd = request.getRequestDispatcher("/WEB-INF/jsp/commentList.jsp");
        rd.include(request, response);
    }
```

### 댓글 쓰기

```java
@Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        request.setCharacterEncoding("UTF-8");

        // AJAX에서는 게시글 ID와 댓글 내용을 각각 "id"와 "comment" 파라미터로 전달합니다.
        String postIdStr = request.getParameter("id");
        int postId = Integer.parseInt(postIdStr);
        String content = request.getParameter("comment");

        // 세션에서 로그인한 사용자 정보를 가져옵니다.
        HttpSession session = request.getSession();
        MemberVO member = (MemberVO) session.getAttribute("member");
        if(member == null) {
            // 로그인하지 않은 경우 적절한 에러 처리
            response.sendError(HttpServletResponse.SC_UNAUTHORIZED, "로그인 필요");
            return;
        }
        String author = member.getId();

        // CommentVO 객체 생성 후 DAO를 통해 DB에 저장
        CommentVO comment = new CommentVO(postId, author, content);
        commentDAO.insertComment(comment);

        // JSON 응답으로 성공 메시지를 보냅니다.
        response.setContentType("application/json;charset=UTF-8");
        PrintWriter out = response.getWriter();
        out.print("{\"status\":\"success\"}");
        out.flush();
    }
```
