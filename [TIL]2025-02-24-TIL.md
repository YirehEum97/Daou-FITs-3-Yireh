```python
from torch.utils.data import Dataset

class TextDataset(Dataset):
    def __init__(self, texts, tokenizer, max_length=128):
        self.texts = texts
        self.tokenizer = tokenizer
        self.max_length = max_length

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        text = self.texts[idx]
        # 토큰화 수행
        inputs = self.tokenizer(text, padding="max_length", truncation=True, max_length=self.max_length, return_tensors="pt")
        # 배치 차원 제거
        inputs = {key: value.squeeze(0) for key, value in inputs.items()}
        return inputs
```

### 유사도 확인

```python
import torch
import torch.nn.functional as F

# 1. 전체 어휘 임베딩 추출
embedding_matrix = model.get_input_embeddings().weight.data  # [vocab_size, embedding_dim]

# 2. 특정 단어의 임베딩 얻기
query_word = "예시"  # 찾고 싶은 단어
query_ids = tokenizer.encode(query_word, add_special_tokens=False)
if len(query_ids) != 1:
    # 단어가 서브워드로 분리될 경우 평균을 내거나 다른 처리가 필요함
    query_embedding = torch.mean(torch.stack([embedding_matrix[id] for id in query_ids]), dim=0)
else:
    query_embedding = embedding_matrix[query_ids[0]]

# 3. 코사인 유사도 계산
# query_embedding: [embedding_dim] -> [1, embedding_dim] 형태로 변경
cosine_similarities = F.cosine_similarity(query_embedding.unsqueeze(0), embedding_matrix, dim=1)

# 4. 유사도 순서대로 정렬 (자기 자신은 가장 유사하므로 제외할 수도 있음)
top_k = 20
similarities, indices = torch.topk(cosine_similarities, top_k + 1)  # 자기자신 포함
# 자기 자신(가장 높은 유사도)를 제외하고 top_k 단어 선택
similar_words = []
for idx in indices[1:top_k+1]:
    token = tokenizer.decode([idx.item()])
    sim = cosine_similarities[idx].item()*100
    similar_words.append((token, sim))

for token, sim in similar_words:
    print(f"{token}: {sim:.2f}")
```
